/**
 * CHỨC NĂNG 1: Tự động gửi Gmail giao việc & Gắn thẳng vào Google Calendar của nhân viên
 * Chạy khi Ban Giám đốc / Tổ trưởng thêm đầu việc mới vào Sheet "05_GIAO_VIEC_BGD"
 */
function syncNewTasksToGmailAndCalendar() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.SHEET_TASKS);
  if (!sheet) return;

  const data = sheet.getDataRange().getValues();
  const calendar = CalendarApp.getDefaultCalendar();

  // Cấu trúc cột: [0] Mã việc | [1] Nội dung việc | [2] Tổ chuyên môn | [3] Họ tên người nhận | [4] Gmail cá nhân | [5] Ngày bắt đầu | [6] Hạn hoàn thành | [7] Mức độ | [8] Đã đồng bộ?
  for (let i = 1; i < data.length; i++) {
    const [taskId, title, dept, staffName, staffEmail, startDate, dueDate, priority, synced] = data[i];

    if (staffEmail && title && synced !== "ĐÃ ĐỒNG BỘ") {
      const deadline = new Date(dueDate);
      const start = startDate ? new Date(startDate) : new Date();

      // 1. Tạo sự kiện trên Google Calendar và mời thẳng Gmail cá nhân của cán bộ
      const eventTitle = `[CN23 - ${dept}] ${title} (${staffName})`;
      const eventDesc = `Nhiệm vụ do Ban Giám đốc Chi nhánh 23 giao.\nNgười phụ trách: ${staffName}\nMức độ: ${priority}\nBáo cáo kết quả tại: ${ss.getUrl()}`;
      
      calendar.createAllDayEvent(eventTitle, deadline, {
        description: eventDesc,
        guests: staffEmail,
        sendInvites: true
      });

      // 2. Gửi Email thông báo giao việc chi tiết
      const subject = `[CHI NHÁNH 23 - GIAO VIỆC] ${title} - Hạn: ${Utilities.formatDate(deadline, "GMT+7", "dd/MM/yyyy")}`;
      const htmlBody = `
        <div style="font-family: Arial, sans-serif; line-height: 1.6; color: #1e293b;">
          <h3 style="color: #1e3a8a;">THÔNG BÁO GIAO NHIỆM VỤ CÔNG TÁC - CHI NHÁNH VPĐKĐĐ SỐ 23</h3>
          <p>Kính gửi đồng chí <strong>${staffName}</strong> (${dept}),</p>
          <p>Ban Giám đốc / Tổ trưởng vừa phân công nhiệm vụ mới cho đồng chí trên Hệ thống Quản lý Công việc:</p>
          <ul>
            <li><strong>Mã công việc:</strong> ${taskId}</li>
            <li><strong>Nội dung nhiệm vụ:</strong> ${title}</li>
            <li><strong>Thời hạn hoàn thành:</strong> <span style="color: #dc2626; font-weight: bold;">${Utilities.formatDate(deadline, "GMT+7", "dd/MM/yyyy")}</span></li>
            <li><strong>Mức độ ưu tiên:</strong> ${priority}</li>
          </ul>
          <p>Sự kiện nhắc hạn đã được tự động thêm vào ứng dụng <strong>Google Calendar</strong> trên điện thoại của đồng chí.</p>
          <p><a href="${ss.getUrl()}" style="background:#1d4ed8;color:white;padding:8px 14px;text-decoration:none;border-radius:6px;">Mở Bảng Theo Dõi Tiến Độ</a></p>
        </div>
      `;

      MailApp.sendEmail({
        to: staffEmail,
        subject: subject,
        htmlBody: htmlBody
      });

      // Đánh dấu đã đồng bộ để không gửi trùng
      sheet.getRange(i + 1, 9).setValue("ĐÃ ĐỒNG BỘ");
    }
  }
}
